      const [x,y,w,hgt]=rect;
      const spot=document.createElement('button');
      spot.className='hotspot'; spot.type='button'; spot.dataset.id=id;
      Object.assign(spot.style,{left:x+'%',top:y+'%',width:w+'%',height:hgt+'%'});
      spot.setAttribute('aria-label', title);
      spot.onclick=()=>{ activate(id,true); openDetailsPopup(title,desc,badge); };
      hs.appendChild(spot);

  const item=document.querySelector('.item[data-id="'+id+'"]');

     Details are shown in the centered popup instead of scrolling. */
}
