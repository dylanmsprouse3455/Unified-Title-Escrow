const hudImages = {"1": "assets/hud1-page-1.webp", "2": "assets/hud1-page-2.webp", "3": "assets/hud1-page-3.webp"};
let currentHudPage = '1';

function renderHUDPage(page){
  currentHudPage = String(page);
  const hudImg = document.getElementById('hudDocImg');
  if(hudImg){
    hudImg.style.display = '';
    hudImg.src = hudImages[currentHudPage];
    hudImg.alt = 'HUD-1 Settlement Statement page ' + currentHudPage;
  }

  document.querySelectorAll('.hud-pagebtn').forEach(btn => {
    btn.classList.toggle('active', btn.dataset.page === currentHudPage);
  });

  const hotspotWrap = document.getElementById('hudHotspots');
  const side = document.getElementById('hudSideContent');
  const quick = document.getElementById('hudQuick');

  if(hotspotWrap) hotspotWrap.innerHTML = '';
  if(side) side.innerHTML = '';

  const map = hudPageMaps[currentHudPage];

  if(quick){
    if(currentHudPage === '1'){
      quick.innerHTML = '<strong>Page 1 — Header + Transaction Summaries</strong><br>Click a mapped section to see where that HUD-1 information comes from in Qualia.';
    } else if(currentHudPage === '2'){
      quick.innerHTML = '<strong>Page 2 — Settlement Charges</strong><br>Sections 700–1300 map the charge categories; Line 1400 totals borrower- and seller-paid settlement charges.';
    } else if(currentHudPage === '3'){
      quick.innerHTML = '<strong>Page 3 — GFE Comparison + Loan Terms</strong><br>The upper half comes from <strong>Disclosures &gt; GFE Comparison</strong>; the lower half comes from <strong>Disclosures &gt; Loan Terms</strong>.';
    }
  }

  if(!map || !map.items || !map.items.length){
    if(side){
      side.innerHTML = '<div class="hud-info-card"><h3>' + (map ? map.heading : 'Not mapped yet') + '</h3><p>The HUD-1 page is available to view. Its clickable Qualia paths will be added from the corresponding Qualia HUD-1 guide.</p></div>';
    }
    return;
  }

  if(side){
    const heading = document.createElement('h3');
    heading.textContent = map.heading;
    side.appendChild(heading);
  }

  map.items.forEach(item => {
    const [x,y,w,h] = item.rect;

    const spot = document.createElement('button');
    spot.type = 'button';
    spot.className = 'hud-hotspot';
    spot.dataset.id = item.id;
    spot.setAttribute('aria-label', item.title);
    Object.assign(spot.style, {
      left: x + '%',
      top: y + '%',
      width: w + '%',
      height: h + '%'
    });
    spot.addEventListener('click', (event) => {
      event.preventDefault();
      event.stopPropagation();
      showHUDDetails(item);
    });
    hotspotWrap.appendChild(spot);

    const card = document.createElement('div');
    card.className = 'hud-section-item';
    card.dataset.id = item.id;
    card.setAttribute('role','button');
    card.setAttribute('tabindex','0');
    card.innerHTML = '<span class="hud-status">QUALIA PATH</span><div class="hud-item-title">' + item.title + '</div>';
    card.addEventListener('click', (event) => {
      event.preventDefault();
      event.stopPropagation();
      showHUDDetails(item);
    });
    card.addEventListener('keydown', (event) => {
      if(event.key === 'Enter' || event.key === ' '){
        event.preventDefault();
        showHUDDetails(item);
      }
    });
    side.appendChild(card);
  });
}

function showHUDDetails(item){
  document.querySelectorAll('.hud-hotspot').forEach(el => {
    el.classList.toggle('active', el.dataset.id === item.id);
  });
  document.querySelectorAll('.hud-section-item').forEach(el => {
    el.classList.toggle('active', el.dataset.id === item.id);
  });

  // Shared centered popup lives at body level so HUD-1 and CD both use it.
  openDetailsPopup(item.title, item.desc, 'HUD-1 • QUALIA');
}

function activateHUDItem(item){
  showHUDDetails(item);
}

document.querySelectorAll('.hud-pagebtn').forEach(btn => {
  btn.addEventListener('click', () => renderHUDPage(btn.dataset.page));
});

      wrap.appendChild(msg);
    }
  });
}


function forceCloseDetailsPopup(){
  const sheet = document.getElementById('iphoneSheet');
  const backdrop = document.getElementById('iphoneBackdrop');
  if(sheet) sheet.classList.remove('open');
  if(backdrop){
    backdrop.classList.remove('open');
    backdrop.setAttribute('aria-hidden','true');
  }
}

function forceCloseTutorial(){
  if(typeof tutorialBackdrop !== 'undefined' && tutorialBackdrop){
    tutorialBackdrop.hidden = true;
  }
}

function clearTransientOverlays(){
  forceCloseDetailsPopup();
  forceCloseTutorial();
}
